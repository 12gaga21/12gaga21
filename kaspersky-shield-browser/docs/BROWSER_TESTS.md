```cpp
// Тест валидации URL
void TestAddressBar::testValidUrl() {
    QStringList validUrls = {
        "https://www.google.com",
        "http://example.com",
        "https://subdomain.example.com"
    };
    
    for (const QString &url : validUrls) {
        QUrl qurl(url);
        QVERIFY(qurl.isValid());
    }
}

// Тест поискового запроса
void TestAddressBar::testSearchQuery() {
    QString query = "how to program";
    QString searchUrl = QString("https://www.google.com/search?q=%1")
                            .arg(QString(query).replace(" ", "+"));
    QVERIFY(searchUrl.contains("google.com/search"));
}
```

```cpp
// Тест полного цикла поиска
void TestBrowserIntegration::testFullSearchCycle() {
    // 1. Ввод поискового запроса
    QString searchQuery = "Qt framework tutorial";
    
    // 2. Преобразование в URL
    QString searchUrl = QString("https://www.google.com/search?q=%1")
                            .arg(QString(searchQuery).replace(" ", "+"));
    
    // 3. Проверка безопасности
    QUrl url(searchUrl);
    auto securityResult = m_securityManager->checkUrl(url);
    QVERIFY(securityResult.safety != UrlSafety::Malicious);
    
    // 4. Загрузка (требует WebEngine)
    QVERIFY(true);
}

// Тест сценария детского профиля
void TestBrowserIntegration::testChildUserScenario() {
    // Активируем детский профиль
    UserProfile childProfile;
    childProfile.profileType = ProfileType::Child;
    m_profileManager->setActiveProfile(childProfile.id);
    
    // Проверяем URL через родительский контроль
    bool allowed = m_parentalControl->isUrlAllowed(url, childProfile.id);
    
    // Логируем попытку доступа
    if (!allowed) {
        qDebug() << "URL заблокирован для детского профиля";
    }
}
```

## 📝 Как добавить новые тесты

### Шаг 1: Создайте тестовый файл

```cpp
#include <QtTest/QtTest>
#include "../include/YourClass.h"

class TestYourClass : public QObject
{
    Q_OBJECT

private slots:
    void testYourFunction();
};

void TestYourClass::testYourFunction() {
    // Ваш тест здесь
    QVERIFY(true);
}

QTEST_MAIN(TestYourClass)
#include "test_yourclass.moc"
```

### Шаг 2: Добавьте в CMakeLists.txt

```cmake
add_executable(test_yourclass
    test_yourclass.cpp
    ../src/YourClass.cpp
)

target_link_libraries(test_yourclass
    Qt6::Core
    Qt6::Test
)

add_test(NAME YourClassTests COMMAND test_yourclass)
```

### Шаг 3: Пересоберите проект

```bash
./scripts/build.sh
```

### Шаг 4: Запустите тесты

```bash
./build_debug/test_yourclass
```

## 🎓 Best Practices тестирования

### 1. Один тест = одна проверка

```cpp
// ❌ Плохо
void testEverything() {
    QVERIFY(condition1);
    QVERIFY(condition2);
    QVERIFY(condition3);
}

// ✅ Хорошо
void testCondition1() {
    QVERIFY(condition1);
}

void testCondition2() {
    QVERIFY(condition2);
}
```

### 2. Используйте setUp и tearDown

```cpp
void init() {
    // Выполняется перед КАЖДЫМ тестом
    m_object = new MyObject();
}

void cleanup() {
    // Выполняется после КАЖДОГО теста
    delete m_object;
}
```

### 3. Информативные сообщения ошибок

```cpp
// ❌ Плохо
QVERIFY(result);

// ✅ Хорошо
QVERIFY2(result, qPrintable(QString("Expected valid result, got: %1").arg(result)));
```
